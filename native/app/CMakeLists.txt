enable_language(C)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
if(ANDROID)
 set(SDL_SHARED ON CACHE BOOL "" FORCE)
 set(SDL_STATIC OFF CACHE BOOL "" FORCE)
endif()
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG f00dc919ca35b9756316e86c4777392de442f9b2 GIT_SHALLOW FALSE GIT_SUBMODULES "")
FetchContent_MakeAvailable(SDL)
set(SDLTTF_VENDORED ON CACHE BOOL "" FORCE)
set(SDLTTF_HARFBUZZ OFF CACHE BOOL "" FORCE)
set(SDLTTF_PLUTOSVG OFF CACHE BOOL "" FORCE)
set(SDLTTF_SAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL_ttf GIT_REPOSITORY https://github.com/libsdl-org/SDL_ttf.git GIT_TAG a1ce3670aec736ecbf0936c43f2f0cc53aa61e5b GIT_SHALLOW FALSE GIT_SUBMODULES external/freetype)
FetchContent_MakeAvailable(SDL_ttf)
set(SDLIMAGE_VENDORED OFF CACHE BOOL "" FORCE)
set(SDLIMAGE_BACKEND_STB ON CACHE BOOL "" FORCE)
set(SDLIMAGE_PNG_LIBPNG OFF CACHE BOOL "" FORCE)
foreach(codec AVIF JXL TIF WEBP)
 set(SDLIMAGE_${codec} OFF CACHE BOOL "" FORCE)
endforeach()
set(SDLIMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDLIMAGE_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL_image GIT_REPOSITORY https://github.com/libsdl-org/SDL_image.git GIT_TAG 0891fc30428b518e7d018677bd6fa95c1a50f4c6 GIT_SHALLOW FALSE GIT_SUBMODULES "")
FetchContent_MakeAvailable(SDL_image)
set(CAPTURE_ASSETS "${CMAKE_BINARY_DIR}/assets")
file(MAKE_DIRECTORY "${CAPTURE_ASSETS}")
file(COPY "${PROJECT_SOURCE_DIR}/native/assets/" DESTINATION "${CAPTURE_ASSETS}")
set(CAPTURE_FONT_PATH "" CACHE FILEPATH "Existing licensed Japanese TTF font (optional)")
option(CAPTURE_FETCH_FONT "Fetch pinned Noto Sans JP for Japanese UI" ON)
if(CAPTURE_FONT_PATH)
 configure_file("${CAPTURE_FONT_PATH}" "${CAPTURE_ASSETS}/fonts/NotoSansJP.ttf" COPYONLY)
elseif(CAPTURE_FETCH_FONT AND NOT EXISTS "${CAPTURE_ASSETS}/fonts/NotoSansJP.ttf")
 file(DOWNLOAD
  "https://raw.githubusercontent.com/google/fonts/295d98a7a0c17c68f1341eaeea354e7960ea70d3/ofl/notosansjp/NotoSansJP%5Bwght%5D.ttf"
  "${CAPTURE_ASSETS}/fonts/NotoSansJP.ttf.download" STATUS font_download TLS_VERIFY ON)
 list(GET font_download 0 font_result)
 if(NOT font_result EQUAL 0)
  message(FATAL_ERROR "Noto Sans JP download failed: ${font_download}. Set CAPTURE_FONT_PATH.")
 endif()
 file(SIZE "${CAPTURE_ASSETS}/fonts/NotoSansJP.ttf.download" font_bytes)
 if(NOT font_bytes EQUAL 9589900)
  message(FATAL_ERROR "Unexpected pinned font size; refusing truncated font")
 endif()
 file(RENAME "${CAPTURE_ASSETS}/fonts/NotoSansJP.ttf.download" "${CAPTURE_ASSETS}/fonts/NotoSansJP.ttf")
endif()
if(NOT EXISTS "${CAPTURE_ASSETS}/fonts/NotoSansJP.ttf")
 message(FATAL_ERROR "Japanese font missing: set CAPTURE_FONT_PATH or CAPTURE_FETCH_FONT=ON")
endif()
set(CAPTURE_SHADER_DIR "" CACHE PATH "Directory containing precompiled .spv shaders")
if(NOT APPLE)
 find_program(CAPTURE_GLSLANG glslangValidator HINTS "$ENV{VULKAN_SDK}/Bin")
 if(NOT CAPTURE_GLSLANG AND NOT CAPTURE_SHADER_DIR)
  message(FATAL_ERROR "Install glslangValidator or set CAPTURE_SHADER_DIR")
 endif()
 set(shader_outputs "")
 foreach(shader scene.vert scene.frag ui.vert ui.frag)
  set(output "${CAPTURE_ASSETS}/shaders/${shader}.spv")
  if(CAPTURE_SHADER_DIR)
   configure_file("${CAPTURE_SHADER_DIR}/${shader}.spv" "${output}" COPYONLY)
  else()
   add_custom_command(OUTPUT "${output}"
    COMMAND "${CAPTURE_GLSLANG}" -V "${PROJECT_SOURCE_DIR}/native/assets/shaders/${shader}" -o "${output}"
    DEPENDS "${PROJECT_SOURCE_DIR}/native/assets/shaders/${shader}" VERBATIM)
  endif()
  list(APPEND shader_outputs "${output}")
 endforeach()
 add_custom_target(capture-shaders DEPENDS ${shader_outputs})
endif()
set(app_sources main.cpp gpu.cpp geometry.cpp ui.cpp)
if(ANDROID)
 add_library(capture-native SHARED ${app_sources})
 set_target_properties(capture-native PROPERTIES OUTPUT_NAME main)
elseif(IOS)
 add_executable(capture-native MACOSX_BUNDLE ${app_sources})
 set_target_properties(capture-native PROPERTIES
  MACOSX_BUNDLE_INFO_PLIST "${PROJECT_SOURCE_DIR}/native/platform/ios/Info.plist"
  MACOSX_BUNDLE_GUI_IDENTIFIER "com.capturebreed.nativeport"
  XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2")
else()
 add_executable(capture-native ${app_sources})
endif()
target_link_libraries(capture-native PRIVATE capture_core SDL3::SDL3 SDL3_ttf::SDL3_ttf SDL3_image::SDL3_image)
target_compile_features(capture-native PRIVATE cxx_std_20)
if(MSVC)
 target_compile_options(capture-native PRIVATE /utf-8 /W4 /fp:precise)
else()
 target_compile_options(capture-native PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
endif()
if(TARGET capture-shaders)
 add_dependencies(capture-native capture-shaders)
endif()
set_target_properties(capture-native PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(NOT ANDROID)
 if(IOS)
  set(asset_destination "$<TARGET_BUNDLE_DIR:capture-native>/assets")
 else()
  set(asset_destination "$<TARGET_FILE_DIR:capture-native>/assets")
 endif()
 add_custom_command(TARGET capture-native POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CAPTURE_ASSETS}" "${asset_destination}" VERBATIM)
 install(TARGETS capture-native RUNTIME DESTINATION . BUNDLE DESTINATION .)
 install(DIRECTORY "${CAPTURE_ASSETS}" DESTINATION .)
endif()
