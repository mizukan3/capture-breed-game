cmake_minimum_required(VERSION 3.24)
project(CaptureBreedNative VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(CAPTURE_BUILD_APP "Build the SDL3 native port workbench" ON)
option(CAPTURE_BUILD_TESTS "Build headless parity and invariant tests" ON)
option(CAPTURE_ENABLE_IPO "Enable interprocedural optimization where supported" ON)
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(nlohmann_json URL https://github.com/nlohmann/json/archive/55f93686c01528224f448c19128836e7df245f72.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP FALSE)
FetchContent_MakeAvailable(nlohmann_json)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/native/data/game.json" CAPTURE_GAME_JSON)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
configure_file(native/src/game_data.hpp.in generated/game_data.hpp @ONLY)
add_library(capture_core STATIC
 native/src/core/data.cpp native/src/core/balance.cpp native/src/core/map.cpp
 native/src/core/combat.cpp native/src/core/camera.cpp native/src/core/checksum.cpp native/src/core/save_store.cpp)
target_include_directories(capture_core PUBLIC native/include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(capture_core PUBLIC nlohmann_json::nlohmann_json)
if(MSVC)
 target_compile_options(capture_core PUBLIC /utf-8 PRIVATE /W4 /permissive- /fp:precise)
else()
 target_compile_options(capture_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
endif()
if(CAPTURE_ENABLE_IPO)
 include(CheckIPOSupported)
 check_ipo_supported(RESULT capture_ipo_supported OUTPUT capture_ipo_error)
 if(capture_ipo_supported)
  set_property(TARGET capture_core PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
 endif()
endif()
if(CAPTURE_BUILD_TESTS AND NOT ANDROID AND NOT IOS)
 enable_testing()
 add_executable(native-map-benchmark native/tools/map_benchmark.cpp)
 target_link_libraries(native-map-benchmark PRIVATE capture_core)
 add_custom_target(profile-map COMMAND $<TARGET_FILE:native-map-benchmark> "${CMAKE_BINARY_DIR}/map-benchmark.json" DEPENDS native-map-benchmark VERBATIM)
 add_executable(native-probe native/tests/probe.cpp)
 target_link_libraries(native-probe PRIVATE capture_core)
 add_executable(core-tests native/tests/core_tests.cpp)
 target_link_libraries(core-tests PRIVATE capture_core)
 add_test(NAME core-invariants COMMAND core-tests)
 add_executable(save-tests native/tests/save_tests.cpp)
 target_link_libraries(save-tests PRIVATE capture_core)
 add_test(NAME save-recovery COMMAND save-tests)
 find_program(CAPTURE_NODE node)
 if(CAPTURE_NODE)
  add_test(NAME html-parity COMMAND "${CAPTURE_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/native/tools/verify_reference.mjs" "$<TARGET_FILE:native-probe>")
 endif()
endif()
if(CAPTURE_BUILD_APP)
 add_subdirectory(native/app)
endif()
